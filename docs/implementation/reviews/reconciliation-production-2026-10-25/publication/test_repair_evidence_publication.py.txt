PUBLISHER = Path(__file__).resolve().parents[1] / "publish_evidence.py"


def extracted_namespace():
    """Load definitions and original 130-path plan without argument parsing/copy."""
    tree = ast.parse(PUBLISHER.read_text(encoding="utf8"))
    nodes = []
    for node in tree.body:
        if isinstance(node, ast.Assign) and any(isinstance(t, ast.Name) and t.id == "parser" for t in node.targets):
            break
        nodes.append(node)
    namespace = {"__file__": str(PUBLISHER)}
    exec(compile(ast.Module(body=nodes, type_ignores=[]), str(PUBLISHER), "exec"), namespace)
    return namespace


def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, sort_keys=True), encoding="utf8")


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def test_original_130_path_plan_and_first42_binding_are_preserved():
    ns = extracted_namespace()
    assert len(ns["sources"]) == len({target for _, target in ns["sources"]}) == 130
    first42 = [(source, target) for source, target in ns["sources"] if target.name == "integrity-first-42files.json"]
    assert len(first42) == 1
    assert first42[0][0] == ns["AUDIT"] / "preparation-only-evidence/initial-42-file-integrity.json"

    pairs = p["ns"]["repair_final_sources"]()

    assert not p["dest"].exists()
    assert not any(source.name in {"research-run.json", "task-bundle.json", "checkpoint.json", "work.sqlite"} for source, _ in pairs)

    assert len([target for source, target in pairs if "original-code" in str(target)]) == 6
    names = {source.name for source, _ in pairs}
    assert {"source-commit-binding-v3.json", "source-commit-binding-v4.json", "generation3-live-recover.stderr.log",

@pytest.mark.parametrize("case", ["same_source", "new_http", "changed_budget", "review_pending", "review_hash",
                                   "body", "pin_body", "financial_claim", "old_summary", "unsafe_source"])

    p = publication
    if case == "same_source":p["actual"]["same_source_cold_and_warm_claim"] = True
    elif case == "new_http":p["actual"]["recovery_new_http"] = 1
    elif case == "changed_budget":p["actual"]["budget_after"] = {"synthetic": 2}
    elif case == "review_pending":p["review"]["status"] = "pending"
    elif case == "body":p["summary"]["questions"][0]["answer"] = {"private": "must not export"}
    elif case == "pin_body":
        write(p["repair"] / "repair-source-pins.json", [{"answers": [{"private": "must not export"}]}])
        p["actual"]["repair_source_pins_sha256"] = sha(p["repair"] / "repair-source-pins.json")
        p["summary"]["repair_source_pins_sha256"] = p["actual"]["repair_source_pins_sha256"]
    elif case == "financial_claim":p["summary"]["financial_accuracy_assessed"] = True
    elif case == "old_summary":p["summary"]["protocol"] = "iqs.shared-production-thin-summary/3.0.0"
    elif case == "unsafe_source":
        p["transition"]["changed_sources"][0]["path"] = str(p["repair"] / "response-body.py")
        write(p["repair"] / "source-transition.json", p["transition"])
        p["actual"]["source_transition_sha256"] = sha(p["repair"] / "source-transition.json")
        p["summary"]["source_repair"]["source_transition_sha256"] = p["actual"]["source_transition_sha256"]

    with pytest.raises(ValueError):p["ns"]["repair_final_sources"]()
    assert not p["dest"].exists()


def history_fixture(tmp_path):
    ns = extracted_namespace()
    root = tmp_path / "synthetic-project"
    dest = root / "docs/production"
    current = root / "runs/current"
    sources, original_rows = [], []
    for index in range(130):
        artifact = "history/evidence-%03d.json" % index
        source, target = current / ("evidence-%03d.json" % index), dest / artifact
        write(source, {"original_metadata_index": index})
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(source.read_bytes())
        original_rows.append(dict(source=str(source.relative_to(root)), artifact=artifact,
                                  sha256=sha(target), bytes=target.stat().st_size))
        sources.append((source, target))
    write(dest / "integrity.json", dict(protocol="iqs.thin-production-evidence/1.0.0", files=original_rows))
    ns.update(ROOT=root, DEST=dest, sources=sources,
              HISTORICAL_TARGETS=frozenset(target for _, target in sources),
              ORIGINAL_ARCHIVE_MANIFEST_SHA256=sha(dest / "integrity.json"),
              options=SimpleNamespace(repair_final=True, final=False))
    return ns, sources, original_rows


def copy_loop_namespace(namespace):
    """Run the real bounded copy loop in fixtures, without final manifest output."""
    tree = ast.parse(PUBLISHER.read_text(encoding="utf8"))
    start = next(i for i, node in enumerate(tree.body) if isinstance(node, ast.Assign)
                 and any(isinstance(target, ast.Name) and target.id == "rows" for target in node.targets))
    end = next(i for i in range(start + 1, len(tree.body)) if isinstance(tree.body[i], ast.Expr)
               and isinstance(tree.body[i].value, ast.Call) and isinstance(tree.body[i].value.func, ast.Attribute)
               and tree.body[i].value.func.attr == "write_text")
    exec(compile(ast.Module(body=tree.body[start:end], type_ignores=[]), str(PUBLISHER), "exec"), namespace)


def test_updated_source_is_revision_while_all_original_130_bytes_and_rows_are_preserved(tmp_path):
    ns, sources, original_rows = history_fixture(tmp_path)
    changed_source, old_target = sources[0]
    original_target_bytes = old_target.read_bytes()
    write(changed_source, dict(resolved_context_root_cause="new thin observation"))
    revision = ns["DEST"] / "history/evidence-000-resolved-after-replay-repair.json"
    ns["sources"] = [*sources, (changed_source, revision)]
    copy_loop_namespace(ns)
    assert old_target.read_bytes() == original_target_bytes
    assert ns["rows"][:130] == original_rows
    assert revision.read_bytes() == changed_source.read_bytes() != original_target_bytes
    assert len(ns["rows"]) == 131


def test_corrupt_old_archive_refuses_before_copying_any_new_revision(tmp_path):
    ns, sources, _ = history_fixture(tmp_path)
    sources[3][1].write_bytes(b"corrupted original archive")
    revision = ns["DEST"] / "new-revision.json"
    ns["sources"] = [*sources, (sources[0][0], revision)]
    with pytest.raises(ValueError, match="historical_archive_bytes_changed"):
        copy_loop_namespace(ns)
    assert not revision.exists()


def test_modified_original_manifest_is_not_silently_treated_as_history(tmp_path):
    ns, sources, _ = history_fixture(tmp_path)
    write(ns["DEST"] / "integrity.json", dict(protocol="iqs.thin-production-evidence/1.0.0", files=[]))
    with pytest.raises(ValueError, match="original_historical_manifest_bytes_changed"):
        ns["preserved_archive_rows"](ns["HISTORICAL_TARGETS"])
