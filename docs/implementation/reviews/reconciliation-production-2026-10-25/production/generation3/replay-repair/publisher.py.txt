def preserved_archive_rows(expected_targets):
    """Verify the existing 130 artifacts and retain original rows, not mutable sources."""
    manifest = DEST / "integrity.json"
    if manifest.is_symlink() or manifest.is_junction():
        raise ValueError("historical_archive_manifest_redirected")
    manifest_raw = manifest.read_bytes()
    if hashlib.sha256(manifest_raw).hexdigest() != ORIGINAL_ARCHIVE_MANIFEST_SHA256:
        raise ValueError("original_historical_manifest_bytes_changed")
    value = json.loads(manifest_raw)
    if (value.get("protocol") != "iqs.thin-production-evidence/1.0.0"
        or len(value.get("files", [])) != 130 or len(expected_targets) != 130):
        raise ValueError("original_130_artifact_manifest_required")
    retained = {}
    for row in value["files"]:
        relative = Path(row["artifact"])
        if relative.is_absolute() or ".." in relative.parts:
            raise ValueError("historical_archive_artifact_path_invalid")
        target = (DEST / relative).absolute()
        if (target in retained or not target.resolve().is_relative_to(DEST.resolve())
            or any(node.is_symlink() or node.is_junction() for node in (target, *target.parents))):
            raise ValueError("historical_archive_artifact_redirected_or_duplicate")
        raw = target.read_bytes()
        if hashlib.sha256(raw).hexdigest() != row["sha256"] or len(raw) != row["bytes"]:
            raise ValueError("historical_archive_bytes_changed")
        retained[target] = row
    if set(retained) != set(expected_targets):
        raise ValueError("historical_archive_target_set_changed")
    return retained

        preserved = target.read_bytes()
        if (hashlib.sha256(preserved).hexdigest() != original_row["sha256"]
            or len(preserved) != original_row["bytes"]):
            raise ValueError("historical_archive_changed_during_publication")
        rows.append(dict(original_row))
        continue
    raw = source.read_bytes()
    # Presence-only key checks; neither names nor values are printed on failure.
    for name, value in os.environ.items():
        if name.endswith("API_KEY") and value and value.encode() in raw:
            raise ValueError("secret_in_selected_evidence")
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, target)
    digest = hashlib.sha256(raw).hexdigest()
    assert hashlib.sha256(target.read_bytes()).hexdigest() == digest
    rows.append(dict(source=os.path.relpath(source, ROOT),
        artifact=str(target.relative_to(DEST)), sha256=digest, bytes=len(raw)))
(DEST / "integrity.json").write_text(json.dumps(dict(
    protocol="iqs.thin-production-evidence/1.0.0", observed_at=datetime.now(timezone.utc).isoformat(),
    actual_real_answer_and_recovery_passed=options.final or options.repair_final,
    accepted_generation=3 if options.repair_final else options.generation if options.final else None,
    financial_accuracy_assessed=False, stockwiki_ack_claimed=False, files=rows,
    **(dict(acceptance_mode="explicit_replay_source_repair", same_source_cold_and_warm_claim=False)
       if options.repair_final else {})), ensure_ascii=False,
    sort_keys=True, indent=2), encoding="utf8")
print(json.dumps(dict(archived_thin_files=len(rows), databases=0, answer_bodies=0)))
